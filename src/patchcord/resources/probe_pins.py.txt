import board
import json
import microcontroller

pins = []
for name in dir(board):
    if name.startswith("_") or name == "board_id":
        continue
    try:
        value = getattr(board, name)
        if isinstance(value, microcontroller.Pin):
            pins.append({"name": name, "identity": id(value)})
    except Exception:
        pass

print("__PATCHCORD_RESULT_V1__" + json.dumps({
    "schema_version": 1,
    "kind": "pins",
    "board_id": board.board_id,
    "pins": pins,
}))
